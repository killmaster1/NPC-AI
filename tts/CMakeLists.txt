cmake_minimum_required(VERSION 3.18)
project(NpcaiKokoro LANGUAGES C CXX)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_CXX_STANDARD 17)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(SHERPA_ONNX_ENABLE_BINARY OFF CACHE BOOL "" FORCE)
set(SHERPA_ONNX_ENABLE_C_API ON CACHE BOOL "" FORCE)
set(SHERPA_ONNX_ENABLE_TTS ON CACHE BOOL "" FORCE)
set(SHERPA_ONNX_ENABLE_JNI OFF CACHE BOOL "" FORCE)
set(SHERPA_ONNX_ENABLE_PORTAUDIO OFF CACHE BOOL "" FORCE)
set(SHERPA_ONNX_ENABLE_WEBSOCKET OFF CACHE BOOL "" FORCE)
set(SHERPA_ONNX_ENABLE_SPEAKER_DIARIZATION OFF CACHE BOOL "" FORCE)
set(SHERPA_ONNX_BUILD_C_API_EXAMPLES OFF CACHE BOOL "" FORCE)
add_subdirectory("${NPC_AI_SHERPA_SOURCE}" sherpa)
add_library(npcai_kokoro SHARED anchor.cpp)
target_link_libraries(npcai_kokoro PRIVATE -Wl,--whole-archive sherpa-onnx-c-api -Wl,--no-whole-archive)
target_link_options(npcai_kokoro PRIVATE -Wl,-z,max-page-size=16384 "-Wl,--version-script=${NPC_AI_SHERPA_SOURCE}/sherpa-onnx/c-api/sherpa-onnx-symbols-c.lds")
